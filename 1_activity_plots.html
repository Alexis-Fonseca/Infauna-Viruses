## ---- Panel B: % active genomes per sample, by treatment (binary) ----
cov <- as.matrix(g[,matched]); cov[is.na(cov)]<-0
gs_active <- rowsum((cov>1)*1,gen)>0
n_tot <- nrow(gs_active)
pa <- data.frame(sample=colnames(gs_active),
                 pct_active=100*colSums(gs_active)/n_tot,
                 Treatment=factor(lab(colnames(gs_active)),levels=ORD))
kp <- kruskal.test(pct_active~Treatment,pa)$p.value
pB <- ggplot(pa,aes(Treatment,pct_active,fill=Treatment))+
  geom_boxplot(width=.6,outlier.shape=NA,alpha=.5)+
  geom_jitter(width=.12,size=2,alpha=.8)+ 
  scale_fill_manual(values=c(Baseline="#8D99AE",HMM="#457B9D",HM="#2A9D8F",LM="#E76F51"))+
  labs(title="B  Transcriptionally active genomes",
       subtitle=sprintf("Kruskal-Wallis p = %.2f (n.s.)",kp),
       y="% of genomes active per sample (>1x)",x=NULL)+
  THEME+theme(legend.position="none")
